cmake_minimum_required(VERSION 3.15)

option(RTYPE_BUILD_SERVER "Build r-type_server (engine core, physics, net)" ON)
option(RTYPE_BUILD_CLIENT "Build r-type_client (engine render, audio, net)" ON)
option(ENGINE_BUILD_SCRIPT "Build engine::script (Lua + sol2)" OFF)
option(ENGINE_BUILD_TOOLS "Build engine::tools (needs render and input)" OFF)

if(RTYPE_BUILD_CLIENT OR ENGINE_BUILD_TOOLS)
    set(RTYPE_NEEDS_SFML ON)
else()
    set(RTYPE_NEEDS_SFML OFF)
endif()
set(ENGINE_BUILD_RENDER ${RTYPE_NEEDS_SFML})
set(ENGINE_BUILD_AUDIO ${RTYPE_BUILD_CLIENT})
set(ENGINE_BUILD_INPUT OFF)
set(ENGINE_BUILD_PHYSICS ${RTYPE_BUILD_SERVER})
set(ENGINE_BUILD_NET ON)

set(VCPKG_MANIFEST_NO_DEFAULT_FEATURES ON)
set(VCPKG_MANIFEST_FEATURES "")
if(RTYPE_BUILD_SERVER)
    list(APPEND VCPKG_MANIFEST_FEATURES server)
endif()
if(RTYPE_BUILD_CLIENT)
    list(APPEND VCPKG_MANIFEST_FEATURES client)
endif()
if(ENGINE_BUILD_SCRIPT)
    list(APPEND VCPKG_MANIFEST_FEATURES script)
endif()
if(ENGINE_BUILD_TOOLS)
    list(APPEND VCPKG_MANIFEST_FEATURES tools)
endif()

project(R-TypeWorkspace VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})

add_subdirectory(engine)
add_subdirectory(games/rtype)
